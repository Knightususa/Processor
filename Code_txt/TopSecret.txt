//Размер Vram по x
push 100
pop [0]

//Размер Vram по y
push 100
pop [1]

//Максимальное количество песчинок
//(сумма этого числа и размера по x должно быть не больше размера RAM)
push 50
pop [2]

//Скорость падения
push 1
pop [3]

//Сдвиг в RAM
push 4
pop cx
//После этого сдвига в RAM идет x потом y координаты песчинки(массив)


//Количество падающих песчинок
push 0
pop dx

//Координата x Дырки
push 50
pop ex

call :InitStolbArr

initw
:loop
    call :randX
    call :Falling
    call :AddPesch
    //call :UpdateStolbArr
    call :MakePicture
    jmp :loop

:randX
    rand 100
    pop ex
    ret

:Falling
    push cx
    pop ax
    :ProverkaEndArr

    //Номер последнего элемента
    push dx
    push 2
    mul
    push cx
    add

    //Нынешний номер элемента
    push ax

    je :EndArrPesch
        push ax + 1
        pop ax
        push [ax]
        push [3]
        add
        pop [ax]
        push ax + 1
        pop ax
        jmp :ProverkaEndArr

    :EndArrPesch
        ret

:AddPesch
//Закидываем новую песчинку в массив
    push dx
    push 2
    mul
    push cx
    add
    pop ax

    push ex
    pop [ax]
    push ax + 1
    pop ax
    push 0
    pop [ax]

    push dx + 1
    pop dx
    ret


:InitStolbArr
    push [2]
    pop ax

    :loopWhile1
        push [0]
        push [2]
        add
        push ax
        je :EndArrStolb
            push 0
            pop [ax]
            push ax + 1
            pop ax
            jmp :loopWhile1

        :EndArrStolb
            ret

:UpdateStolbArr
    push cx
    pop ax

    :loopWhile2
        push dx
        push 2
        mul
        push cx
        add
        push ax
        je :EndArrStolb1
            push [ax]
            push [2]
            add
            pop bx

            //высота столба по этой x 
            push [bx]
            push ax + 1
            pop ax
            push [ax]
            add
            //теперь в стеке сумма высоты столба и координаты y
            push [1]
            


        :EndArrStolb1
            ret

:MakePicture
    push 0
    pop bx

    push [2]
    pop ax

    :loopWhile3
        push [0]
        push [2]
        add
        push ax
        je :EndArrStolb2
            push [ax]
            push ax + 1
            pop ax
            push [ax]
            push ax + 1
            pop ax

            push [0]
            mul
            add
            pop bx
            //Номер элемента в Vram
            push 255
            pop {bx}
            jmp :loopWhile3

        :EndArrStolb
            draw
            delay(1)
            ret
